# The UTF-8 switch (2026-10-31) refuses LEN of a STR.  The @new below
# says Text.BLen (s) where it said LEN (s).
@name the time bound: a search that runs an anchored match from every start
@old
PROCEDURE Search (re: PTR Re ; RO s: STR ; from: I64) : OPT PTR Found =
BEGIN
  RETURN Run (re, s, from, FALSE, FALSE, FALSE)
END Search ;
@new
PROCEDURE Search (re: PTR Re ; RO s: STR ; from: I64) : OPT PTR Found =
VAR i : I64 ; r : OPT PTR Found ;
BEGIN
  FOR i := from TO Text.BLen (s) DO
    r := Run (re, s, i, TRUE, FALSE, FALSE) ;
    IF r IS SOME m THEN RETURN r END
  END ;
  RETURN NONE
END Search ;
@end
