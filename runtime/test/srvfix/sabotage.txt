# runtime/test/Sabotage.m9 srvfix reads this list.  Each entry is one
# small edit to a COPY of a file.  The edit breaks one property the
# httpserver gate (srvfix.py) claims to hold, so the gate must FAIL on it.
# docs/httpserver-plan.md, stages 1 to 3.
#
# The form, one entry:
#
#   @name what the edit breaks
#   @file corpus/Http.m9      (optional; corpus/HttpServer.m9 without it)
#   @old
#   the text to replace: it must occur in the file exactly once
#   @new
#   the text put in its place
#   @end
#
# A text is the lines between its two markers, joined by newlines, taken
# exactly: leading spaces count, and no escapes exist.  A text that ends
# in a newline has one empty line before the next marker.  An empty text
# has no lines.  A line beginning with # is a comment only outside an
# entry.  A file under corpus/ is edited in the copied library, any other
# in the copied runtime (by its base name).
#
# Not here: the directory check in SendFile.  Its removal is visible on
# macOS only, where a directory has a size.  On Linux the size query
# fails first and the gate answers 404 either way (measured).

# Since the UTF-8 switch an error line leaves in ONE fwrite, which the C
# library locks itself, so removing M9_LOCK alone broke nothing and this
# entry stayed green (ct121, 2026-10-31).  It now breaks the property
# where it lives: the line goes out an octet at a time, unlocked.
@name stderr not locked for a whole line
@file runtime/m9rt.c
@old
  M9_LOCK (stderr);
  if (n > 0) fwrite (buf, 1, n, stderr);
  fflush (stderr);
  M9_UNLOCK (stderr);
@new
  for (size_t k = 0; k < n; k++) fputc (((const unsigned char *) buf)[k], stderr);
  fflush (stderr);
@end
